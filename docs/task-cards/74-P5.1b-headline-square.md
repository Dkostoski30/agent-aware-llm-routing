Run (the user, laptop idle, Windows' AC standby off, `LightingService` stopped as admin first; both squares in one
block, about 9.5 h; up to 3 tries per series, each failed one's partial file moved aside first, since `make experiment`
refuses a file's leftovers; output in `~/p5-squares.log`):
```
kill %1 2>/dev/null
sudo -v
while sudo -n -v; do sleep 60; done &
ls_ok() { for t in 1 2 3; do s=$(sc.exe query LightingService 2>/dev/null); l=$(tasklist.exe 2>/dev/null); if echo "$l" | grep -q "System Idle Process"; then echo "$s" | grep -q STOPPED && ! echo "$l" | grep -qi lm_support && return 0; return 1; fi; sleep 5; done; return 1; }
tidy() { d=~/thesis/experiments/results; t=$d/$1-failed-$(date +%H%M); for st in $(ls $d | grep -oE "^$1-[a-d]-[0-9a-f]{12}" | sort -u); do [ -f $d/$st.meta.json ] || { mkdir -p $t; mv $d/$st-* $t/; echo "tidy: moved $st's partial results to $t"; }; done; kubectl scale deployment/vllm --replicas=2; kubectl rollout status deployment/vllm --timeout=900s; }
run() { for try in 1 2 3; do make -C ~/thesis experiment-series NAME=$1 2>&1 | tee -a ~/p5-squares.log; [ "${PIPESTATUS[0]}" -eq 0 ] && return 0; echo "series $1: try $try failed at $(date +%H:%M), tidying" | tee -a ~/p5-squares.log; tidy $1 2>&1 | tee -a ~/p5-squares.log; done; return 1; }
go=no; ls_ok && go=yes; echo "check: LightingService stopped and LM_Support gone: $go"
[ "$go" = yes ] && run p5-1b
[ "$go" = yes ] && run p5-1c
kill %1
```
